1// Read-only source comments v1. Provider IDs and cursors belong to each source.
2use crate::source_protocol::{direct, feature, raw_string};
3
4pub const OPERATION: &str = "get_comments";
5
6pub struct Request<'a> {
7    pub manga_id: &'a [u8],
8    pub chapter_id: Option<&'a [u8]>,
9    pub cursor: Option<&'a [u8]>,
10}
11
12pub fn request(input: &[u8]) -> Result<Request<'_>, &'static str> {
13    if !feature(input, b"source_comments_v1") {
14        return Err("host_feature_required");
15    }
16    let args = direct(input, b"args")?.ok_or("invalid_request")?;
17    let manga_id = raw_string(args, b"mangaId")?
18        .filter(|id| !id.is_empty())
19        .ok_or("invalid_request")?;
20    let chapter_id = raw_string(args, b"chapterId")?;
21    match (raw_string(args, b"scope")?, chapter_id) {
22        (Some(b"manga"), None) => {}
23        (Some(b"chapter"), Some(id)) if !id.is_empty() => {}
24        _ => return Err("invalid_request"),
25    }
26    Ok(Request {
27        manga_id,
28        chapter_id,
29        cursor: raw_string(args, b"cursor")?,
30    })
31}

33pub fn position(cursor: Option<&[u8]>) -> Result<Option<usize>, &'static str> {
34    cursor
35        .map(|raw| {
36            if raw.is_empty() || !raw.iter().all(u8::is_ascii_digit) {
37                return Err("invalid_request");
38            }
39            core::str::from_utf8(raw)
40                .ok()
41                .and_then(|s| s.parse().ok())
42                .ok_or("invalid_request")
43        })
44        .transpose()
45}

47// Convert a Unix millisecond timestamp to ISO 8601 for the host.
48pub fn utc_iso_millis(raw: &[u8], out: &mut [u8; 20]) -> Option<usize> {
49    let stamp = core::str::from_utf8(raw).ok()?.parse::<u64>().ok()?;
50    let seconds = stamp / 1000;
51    let days = seconds / 86_400;
52    if days > 2_932_896 {
53        return None;
54    }
55    let z = days + 719_468;
56    let era = z / 146_097;
57    let doe = z % 146_097;
58    let yoe = (doe - doe / 1_460 + doe / 36_524 - doe / 146_096) / 365;
59    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
60    let mp = (5 * doy + 2) / 153;
61    let day = doy - (153 * mp + 2) / 5 + 1;
62    let month = if mp < 10 { mp + 3 } else { mp - 9 };
63    let year = era * 400 + yoe + u64::from(month <= 2);
64    *out = *b"0000-00-00T00:00:00Z";
65    for (start, width, mut value) in [
66        (0, 4, year),
67        (5, 2, month),
68        (8, 2, day),
69        (11, 2, (seconds / 3600) % 24),
70        (14, 2, (seconds / 60) % 60),
71        (17, 2, seconds % 60),
72    ] {
73        for index in (start..start + width).rev() {
74            out[index] = b'0' + (value % 10) as u8;
75            value /= 10;
76        }
77    }
78    Some(out.len())
79}
